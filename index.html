import requests import json import os import time from datetime import datetime def assign_category(title): title = title.lower() technology = ["ai", "software", "tech", "code", "computer", "data", "cloud", "api", "cpu", "llm"] world_news = ["war", "government", "country", "president", "election", "climate", "attack", "global"] sports = ["nfl", "nba", "f1", "fes", "sports", "game", "team", "players", "league", "championship"] science = ["research", "study", "space", "physics", "biology", "discovery", "nasa", "genomy"] entertainment = ["movie", "film", "music", "netflix", "game", "book", "show", "awards", "streaming"] if any(word in title for word in technology): return "Technology" elif any(word in title for word in world_news): return "World News" elif any(word in title for word in sports): return "Sports" elif any(word in title for word in science): return "Science" elif any(word in title for word in entertainment): return "Entertainment" else: return "Other" top_stories_url = "https://hacker-news.firebaseio.com/v0/topstories.json" headers = { "User-Agent": "TrendPlus 0.1.0" } try: response = requests.get( top_stories_url, headers=headers, timeout=10 ) response.raise_for_status() story_ids = response.json()[:500] except requests.RequestException as error: print("Failed to fetch top stories:", error) story_ids = [] stories = [] for story_id in story_ids: item_url = ( f"https://hacker-news.firebaseio.com/v0/item/{story_id}.json" ) try: response = requests.get( item_url, headers=headers, timeout=10 ) response.raise_for_status() item = response.json() if not item or item.get("type") != "story": continue title = item.get("title", "") story = { "post_id": item.get("id"), "title": title, "description": item.get("text", ""), "category": assign_category(title), "score": item.get("score", 0), "num_comments": item.get("descendants", 0), "author": item.get("by", ""), "collected_at": datetime.now().isoformat() } stories.append(story) except requests.RequestException as error: print(f"Failed to fetch story {story_id}: {error}") continue os.makedirs("Data", exist_ok=True) date_string = datetime.now().strftime("%Y-%m-%d") file_path = f"Data/data_{date_string}.json" with open(file_path, "w", encoding="utf-8") as file: json.dump(stories, file, indent=4, ensure_ascii=False) print(f"Collected {len(stories)} stories, saved to {file_path}")
